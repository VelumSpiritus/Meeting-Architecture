import time
from dataclasses import dataclass, field
from typing import Dict, List, Optional

@dataclass(frozen=True)
class Account:
    """Permanent system identity with explicit operational roles."""
    account_id: str
    public_key: str
    role: str  # "CENTRAL_BANK", "COMMERCIAL_BANK", "EDGE_NODE"

@dataclass(frozen=True)
class Token:
    """Cryptographic abstraction of a tokenized asset state."""
    token_id: str
    asset_type: str  # "CBDC_USD", "TOKENIZED_DEPOSIT"
    issuer_id: str

@dataclass
class Transaction:
    """Signed ledger operation that directly mutates state cache."""
    tx_id: str
    sender_id: str
    receiver_id: str
    token_id: str
    amount: float
    timestamp: float = field(default_factory=time.time)
    signature: Optional[str] = None

class UniversalLedgerCache:
    """Localized, high-throughput execution environment (The Ledger Cache)."""
    def __init__(self):
        self.accounts: Dict[str, Account] = {}
        self.balances: Dict[str, Dict[str, float]] = {}
        self.transaction_history: List[Transaction] = []
        self.state_root: str = self._calculate_genesis_hash()

    def _calculate_genesis_hash(self) -> str:
        return hashlib.sha256(b"GENESIS_STATE_BLOCK_0").hexdigest()

    def register_identity(self, account: Account):
        """Permanent identity commit to the ledger layer."""
        if account.account_id not in self.accounts:
            self.accounts[account.account_id] = account
            self.balances[account.account_id] = {}

    def mint_asset(self, operator_id: str, token: Token, amount: float):
        """Restricted: only CENTRAL_BANK role can generate liquidity."""
        operator = self.accounts.get(operator_id)
        if not operator or operator.role != "CENTRAL_BANK":
            raise PermissionError(
                "Security Violation: Unauthorized Mint Attempt Detected.")

        self.register_identity(
            Account(token.issuer_id, "SYSTEM_KEY", "CENTRAL_BANK"))
        current = self.balances[token.issuer_id].get(token.token_id, 0.0)
        self.balances[token.issuer_id][token.token_id] = current + amount

    def execute_atomic_settlement(self, tx: Transaction) -> bool:
        """State transition inside the local edge memory cache."""
        # 1. Validate systemic identity presence
        if tx.sender_id not in self.accounts \
           or tx.receiver_id not in self.accounts:
            return False

        # 2. Check localized cache liquidity
        sender_balance = self.balances[tx.sender_id].get(tx.token_id, 0.0)
        if sender_balance < tx.amount:
            return False  # No state corruption or overdraw

        # 3. Mutate the Ledger Cache State — instantly
        self.balances[tx.sender_id][tx.token_id] = sender_balance - tx.amount
        receiver_bal = self.balances[tx.receiver_id].get(tx.token_id, 0.0)
        self.balances[tx.receiver_id][tx.token_id] = receiver_bal + tx.amount

        # 4. Commit to immutable log, recalculate state hash
        self.transaction_history.append(tx)
        self._update_state_root(tx)
        return True

    def _update_state_root(self, tx: Transaction):
        """Merkle-style root hash — state integrity verification."""
        tx_str = f"{tx.tx_id}{tx.sender_id}{tx.receiver_id}" \
                 f"{tx.amount}{self.state_root}"
        self.state_root = hashlib.sha256(tx_str.encode('utf-8')).hexdigest()
